# skimr v0.2.0 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship skimr v0.2.0 as the RAG-prep primitive — summary + structured enrichments (stats, outline, metadata, phrases, correlated_facts) in a single sub-250ms call with cross-language (Python + Rust) parity.

**Architecture:** Big-bang release. `summarize()` gains a mode selector (`default`/`coverage`/`legacy`) and an `attach=[...]` kwarg; new `skimr.extract.*` namespace houses structured-data primitives; NER ships as an optional `skimr[ner]` extra (Python-only). Rust port maintains byte-identity for the stdlib path; NER entities left empty in Rust by design.

**Tech Stack:** Python 3.12+ (stdlib core; optional spaCy for NER), Rust edition 2024 MSRV 1.85 (regex-only runtime dep), existing fixture harness, pytest, cargo test, pytest-cov, vLLM-hosted Qwen for A4 judge.

### Step 3: Implement `src/skimr/_headings.py`

```python
"""Shared heading detection.

Any sentence matching any of these patterns is considered a heading and is
dropped from candidate selection in mode='default'. Also used by
extract.outline to discover section names.
"""
import re

# Markdown ATX-style heading: one or more # followed by space and text
_MD_HEADING_RE = re.compile(r"^\s*#+\s+.+$")

# ALL-CAPS short line: 4-30 chars of A-Z/space/colon, no lowercase
_ALLCAPS_RE = re.compile(r"^\s*[A-Z][A-Z\s]{3,28}:?\s*$")

# Short label ending in colon (<=30 chars including the colon)
_SHORT_LABEL_RE = re.compile(r"^\s*.{1,30}:\s*$")


def is_heading(sentence: str) -> bool:
    """True when `sentence` matches any heading pattern."""
    if not sentence.strip():
        return False
    if _MD_HEADING_RE.match(sentence):
        return True
    if _ALLCAPS_RE.match(sentence):
        return True
    if _SHORT_LABEL_RE.match(sentence):
        return True
    # Fewer than 4 content-word tokens (rough "title-like" filter).
    toks = [t for t in re.findall(r"[A-Za-z]{3,}", sentence)]
    if len(toks) < 4:
        return True
    return False


def heading_name(sentence: str) -> str | None:
    """Extract the name portion of a heading, or None if not a heading."""
    s = sentence.strip()
    if not s:
        return None
    # Strip markdown #'s and trailing colon.
    s = re.sub(r"^#+\s+", "", s)
    s = s.rstrip(":").strip()
    if not s:
        return None
    return s
```

```python
def _composite_score_default(sentences, tokenized, section_map):
    """Mode='default' scorer — v0.2.0 C1 tweaks.

    - Heading filter: sentences flagged is_heading() get score -inf (dropped).
    - Cue-phrase boost: +2.0 when sentence starts with a cue phrase.
    - Digit bonus: +0.3 when sentence contains a digit.
    - Section-position weighting: TF-IDF component *= 1.3 when inside a
      boosted section ({discussion, conclusion, held, ...}).

    section_map: index -> lowercase section-name string (or "").
    """
    # Compute baseline TF-IDF + position + length scores (reuse legacy path)
    base = _composite_score_legacy(sentences, tokenized)
    scores = []
    for i, sentence in enumerate(sentences):
        if is_heading(sentence):
            scores.append(float("-inf"))
            continue
        tfidf_part, pos_part, len_part = base[i]  # legacy returns 3-tuple; refactor below
        if section_map.get(i, "") in _SECTION_BOOST_HEADINGS:
            tfidf_part *= 1.3
        score = 0.60 * tfidf_part + 0.25 * pos_part + 0.15 * len_part
        if _CUE_PHRASE_RE.match(sentence):
            score += 2.0
        if _DIGIT_RE.search(sentence):
            score += 0.3
        scores.append(score)
    return scores
```

//! Shared heading detection — mirrors src/skimr/_headings.py.

use regex::Regex;
use std::sync::OnceLock;

fn md_heading_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^\s*#+\s+.+$").expect("static regex"))
}

fn allcaps_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^\s*[A-Z][A-Z\s]{3,28}:?\s*$").expect("static regex"))
}

fn short_label_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^\s*.{1,30}:\s*$").expect("static regex"))
}

fn content_word_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"[A-Za-z]{3,}").expect("static regex"))
}

#[must_use]
pub fn is_heading(sentence: &str) -> bool {
    let trimmed = sentence.trim();
    if trimmed.is_empty() {
        return false;
    }
    if md_heading_re().is_match(sentence) {
        return true;
    }
    if allcaps_re().is_match(sentence) {
        return true;
    }
    if short_label_re().is_match(sentence) {
        return true;
    }
    let toks: Vec<_> = content_word_re().find_iter(sentence).collect();
    toks.len() < 4
}

#[must_use]
pub fn heading_name(sentence: &str) -> Option<String> {
    let s = sentence.trim();
    if s.is_empty() {
        return None;
    }
    // Strip leading "#+ " prefix
    let stripped = Regex::new(r"^#+\s+").expect("static regex").replace(s, "");
    let stripped = stripped.trim_end_matches(':').trim();
    if stripped.is_empty() {
        None
    } else {
        Some(stripped.to_string())
    }
}
```

```python
"""Coverage-constrained selector. Activated via summarize(mode='coverage').

Picks the highest-scoring sentence from each paragraph (paragraphs are
blank-line-delimited; fewer-than-20-char paragraphs are skipped). After a
one-per-paragraph pass, the next-best ungrabbed sentences fill remaining
budget. Reorder by original document position.
"""
import re

from ._headings import is_heading
from .tfidf import _composite_score_parts, _split_sentences, _tokenize


_PARA_SPLIT_RE = re.compile(r"\n\s*\n+")


def _split_paragraphs(text: str) -> list[str]:
    """Return paragraphs of at least 20 chars after strip."""
    return [p.strip() for p in _PARA_SPLIT_RE.split(text) if len(p.strip()) >= 20]


def _paragraph_of(text: str, sentence: str) -> int:
    """Return the index of the paragraph containing `sentence`, or -1."""
    paras = _split_paragraphs(text)
    for i, p in enumerate(paras):
        if sentence in p:
            return i
    return -1


def summarize_coverage(text: str, max_length: int = 500) -> str:
    """C2 coverage selector."""
    sentences = _split_sentences(text)
    if not sentences:
        return text
    tokenized = [_tokenize(s) for s in sentences]
    parts = _composite_score_parts(sentences, tokenized)
    # Combined score, same weights as default mode, but no cue/digit boosts
    # — coverage uses raw TF-IDF/position/length only.
    scores = [0.60 * t + 0.25 * p + 0.15 * l for (t, p, l) in parts]

    # Map sentence -> paragraph index
    paras = _split_paragraphs(text)
    sent_to_para: list[int] = []
    for s in sentences:
        idx = -1
        for i, p in enumerate(paras):
            if s in p:
                idx = i
                break
        sent_to_para.append(idx)

    # For each paragraph, pick the top-scoring non-heading sentence
    best_per_para: dict[int, tuple[float, int]] = {}
    for i, s in enumerate(sentences):
        if is_heading(s):
            continue
        para_idx = sent_to_para[i]
        if para_idx < 0:
            continue
        if para_idx not in best_per_para or scores[i] > best_per_para[para_idx][0]:
            best_per_para[para_idx] = (scores[i], i)

    selected = sorted({idx for (_, idx) in best_per_para.values()})

    # Fill remaining budget with next-best globally, skipping already-selected and headings.
    remaining = [(scores[i], i) for i in range(len(sentences))
                 if i not in selected and not is_heading(sentences[i])]
    remaining.sort(key=lambda x: -x[0])

    def total_chars(indices: list[int]) -> int:
        return sum(len(sentences[i]) + 1 for i in indices)  # +1 for newline join

    budget_used = total_chars(selected)
    for _, i in remaining:
        if budget_used + len(sentences[i]) + 1 <= max_length:
            selected.append(i)
            budget_used += len(sentences[i]) + 1
        if budget_used >= max_length:
            break

    selected.sort()
    return "\n".join(sentences[i] for i in selected)
```

//! Coverage-constrained selector. Mirrors src/skimr/coverage.py.

use crate::headings::is_heading;
use crate::sentences::split_sentences;
use crate::tfidf::{composite_score_parts, greedy_tokenize};
use regex::Regex;
use std::sync::OnceLock;

fn para_split_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"\n\s*\n+").expect("static regex"))
}

fn split_paragraphs(text: &str) -> Vec<String> {
    para_split_re()
        .split(text)
        .map(|p| p.trim().to_string())
        .filter(|p| p.chars().count() >= 20)
        .collect()
}

#[must_use]
pub fn summarize_coverage(text: &str, max_length: usize) -> String {
    let sentences = split_sentences(text);
    if sentences.is_empty() {
        return text.to_string();
    }
    let tokenized: Vec<Vec<String>> = sentences.iter().map(|s| greedy_tokenize(s)).collect();
    let parts = composite_score_parts(&sentences, &tokenized);
    let scores: Vec<f64> = parts
        .iter()
        .map(|(t, p, l)| 0.60 * t + 0.25 * p + 0.15 * l)
        .collect();

    let paragraphs = split_paragraphs(text);
    let sent_to_para: Vec<i32> = sentences
        .iter()
        .map(|s| {
            for (i, p) in paragraphs.iter().enumerate() {
                if p.contains(s.as_str()) {
                    return i as i32;
                }
            }
            -1
        })
        .collect();

    // Best (score, index) per paragraph
    let mut best_per_para: std::collections::BTreeMap<i32, (f64, usize)> =
        std::collections::BTreeMap::new();
    for (i, s) in sentences.iter().enumerate() {
        if is_heading(s) {
            continue;
        }
        let pidx = sent_to_para[i];
        if pidx < 0 {
            continue;
        }
        let entry = best_per_para.entry(pidx).or_insert((f64::NEG_INFINITY, i));
        if scores[i] > entry.0 {
            *entry = (scores[i], i);
        }
    }

    let mut selected: Vec<usize> = best_per_para.values().map(|(_, i)| *i).collect();
    selected.sort_unstable();
    selected.dedup();

    // Fill remaining budget
    let mut remaining: Vec<(f64, usize)> = (0..sentences.len())
        .filter(|i| !selected.contains(i) && !is_heading(&sentences[*i]))
        .map(|i| (scores[i], i))
        .collect();
    remaining.sort_by(|a, b| b.0.partial_cmp(&a.0).expect("no NaN"));

    let total_chars = |sel: &[usize]| -> usize {
        sel.iter().map(|i| sentences[*i].len() + 1).sum::<usize>()
    };

    let mut used = total_chars(&selected);
    for (_, i) in remaining {
        let cost = sentences[i].len() + 1;
        if used + cost <= max_length {
            selected.push(i);
            used += cost;
        }
        if used >= max_length {
            break;
        }
    }

    selected.sort_unstable();
    selected.iter().map(|i| sentences[*i].clone()).collect::<Vec<_>>().join("\n")
}
```

```python
"""Hierarchical outline extractor."""
import re
from dataclasses import replace
from .._headings import is_heading, heading_name
from .._types import Section
from ..tfidf import _split_sentences, _tokenize, _composite_score_parts


_MD_DEPTH_RE = re.compile(r"^\s*(#+)\s+")


def _md_depth(heading_line: str) -> int:
    """Return markdown heading depth (# = 1, ## = 2, etc.), or 1 for non-markdown."""
    m = _MD_DEPTH_RE.match(heading_line)
    if m:
        return len(m.group(1))
    return 1


def outline(text: str) -> tuple[Section, ...]:
    """Extract sections: heading + top-scoring non-heading sentence within."""
    sentences = _split_sentences(text)
    if not sentences:
        return ()

    # Score all sentences once so we can pick the best per section.
    tokenized = [_tokenize(s) for s in sentences]
    parts = _composite_score_parts(sentences, tokenized)
    scores = [0.60 * t + 0.25 * p + 0.15 * l for (t, p, l) in parts]

    # Walk sentences; record heading positions and group subsequent
    # non-heading sentences under each heading.
    sections: list[tuple[int, int, str, list[int]]] = []
    # tuple: (heading_index, depth, name, [body_sentence_indices])
    current: tuple[int, int, str, list[int]] | None = None
    for i, s in enumerate(sentences):
        if is_heading(s):
            if current is not None:
                sections.append(current)
            name = heading_name(s) or ""
            current = (i, _md_depth(s), name, [])
        else:
            if current is not None:
                current[3].append(i)
    if current is not None:
        sections.append(current)

    # For each section, pick the highest-scoring body sentence.
    out: list[Section] = []
    for _, depth, name, body in sections:
        if not body or not name:
            continue
        best_idx = max(body, key=lambda i: scores[i])
        out.append(Section(
            depth=depth,
            name=name,
            representative_sentence=sentences[best_idx],
        ))
    return tuple(out)
```

fn md_depth_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^\s*(#+)\s+").expect("static regex"))
}

fn md_depth(heading_line: &str) -> usize {
    if let Some(caps) = md_depth_re().captures(heading_line) {
        return caps.get(1).map_or(1, |m| m.as_str().len());
    }
    1
}

#[must_use]
pub fn outline(text: &str) -> Vec<Section> {
    let sentences = split_sentences(text);
    if sentences.is_empty() {
        return Vec::new();
    }
    let tokenized: Vec<Vec<String>> = sentences.iter().map(|s| greedy_tokenize(s)).collect();
    let parts = composite_score_parts(&sentences, &tokenized);
    let scores: Vec<f64> = parts
        .iter()
        .map(|(t, p, l)| 0.60 * t + 0.25 * p + 0.15 * l)
        .collect();

    // Group sentences under headings
    struct Sect {
        depth: usize,
        name: String,
        body: Vec<usize>,
    }
    let mut sections: Vec<Sect> = Vec::new();
    let mut current: Option<Sect> = None;
    for (i, s) in sentences.iter().enumerate() {
        if is_heading(s) {
            if let Some(cur) = current.take() {
                sections.push(cur);
            }
            let name = heading_name(s).unwrap_or_default();
            current = Some(Sect { depth: md_depth(s), name, body: Vec::new() });
        } else if let Some(cur) = current.as_mut() {
            cur.body.push(i);
        }
    }
    if let Some(cur) = current {
        sections.push(cur);
    }

    let mut out = Vec::with_capacity(sections.len());
    for sect in sections {
        if sect.body.is_empty() || sect.name.is_empty() {
            continue;
        }
        let best_idx = *sect
            .body
            .iter()
            .max_by(|a, b| scores[**a].partial_cmp(&scores[**b]).expect("no NaN"))
            .expect("non-empty");
        out.push(Section {
            depth: sect.depth,
            name: sect.name,
            representative_sentence: sentences[best_idx].clone(),
        });
    }
    out
}
```

import re
from .._types import Stat
from ..tfidf import _split_sentences


_MONEY_RE = re.compile(
    r"(?P<value>\$\d[\d,]*(?:\.\d+)?[KMB]?)"
    r"|(?P<value2>\d[\d,]*(?:\.\d+)?)\s*(?P<ccy>dollars?|USD|EUR|GBP|JPY|CHF)",
    re.IGNORECASE,
)

_PERCENT_RE = re.compile(
    r"(?P<value>\d+(?:\.\d+)?)\s*(?P<unit>%|percent)",
    re.IGNORECASE,
)

_DATE_RE = re.compile(
    r"(?P<iso>\d{4}-\d{2}-\d{2})"
    r"|(?P<us>\d{1,2}/\d{1,2}/\d{2,4})",
)

_DURATION_RE = re.compile(
    r"(?P<value>\d+)\s*(?P<unit>seconds?|minutes?|hours?|days?|weeks?|months?|years?)",
    re.IGNORECASE,
)

_COUNT_RE = re.compile(
    r"(?P<value>\d[\d,]*)\s*(?P<unit>events?|users?|customers?|requests?"
    r"|per second|per minute|per hour|qps|rps|chunks?)",
    re.IGNORECASE,
)


def _ctx_phrase(sentence: str, start: int, end: int, window: int = 25) -> str:
    """Return `sentence[start-window:end+window]`, clipped to token boundaries."""
    l = max(0, start - window)
    r = min(len(sentence), end + window)
    return sentence[l:r].strip()


def stats(text: str) -> tuple[Stat, ...]:
    """Extract money, percent, date, duration, count facts with context."""
    out: list[Stat] = []
    for sent in _split_sentences(text):
        for m in _MONEY_RE.finditer(sent):
            value = m.group("value") or m.group("value2")
            out.append(Stat(
                value=value,
                unit="usd",
                phrase=_ctx_phrase(sent, m.start(), m.end()),
                context_sentence=sent,
                stat_type="money",
            ))
        for m in _PERCENT_RE.finditer(sent):
            out.append(Stat(
                value=m.group("value"),
                unit="percent",
                phrase=_ctx_phrase(sent, m.start(), m.end()),
                context_sentence=sent,
                stat_type="percent",
            ))
        for m in _DATE_RE.finditer(sent):
            value = m.group("iso") or m.group("us")
            out.append(Stat(
                value=value,
                unit="date",
                phrase=_ctx_phrase(sent, m.start(), m.end()),
                context_sentence=sent,
                stat_type="date",
            ))
        for m in _DURATION_RE.finditer(sent):
            out.append(Stat(
                value=f"{m.group('value')} {m.group('unit').lower()}",
                unit=m.group("unit").lower().rstrip("s"),
                phrase=_ctx_phrase(sent, m.start(), m.end()),
                context_sentence=sent,
                stat_type="duration",
            ))
        for m in _COUNT_RE.finditer(sent):
            out.append(Stat(
                value=m.group("value"),
                unit=m.group("unit").lower(),
                phrase=_ctx_phrase(sent, m.start(), m.end()),
                context_sentence=sent,
                stat_type="count",
            ))
    return tuple(out)

fn ctx(sent: &str, start: usize, end: usize) -> String {
    let window = 25;
    let l = start.saturating_sub(window);
    let r = (end + window).min(sent.len());
    sent[l..r].trim().to_string()
}

pub fn stats(text: &str) -> Vec<Stat> {

            let value = caps
                .name("a")
                .or_else(|| caps.name("b"))
                .map(|x| x.as_str().to_string())
                .unwrap_or_default();

                value,
                unit: "usd".into(),
                phrase: ctx(&sent, m.start(), m.end()),
                context_sentence: sent.clone(),
                stat_type: "money".into(),
            });
        }
        for caps in percent_re().captures_iter(&sent) {
            let m = caps.get(0).expect("match");
            out.push(Stat {
                value: caps["v"].to_string(),
                unit: "percent".into(),
                phrase: ctx(&sent, m.start(), m.end()),
                context_sentence: sent.clone(),
                stat_type: "percent".into(),
            });
        }
        for caps in date_re().captures_iter(&sent) {
            let m = caps.get(0).expect("match");
            let value = caps
                .name("iso")
                .or_else(|| caps.name("us"))
                .map(|x| x.as_str().to_string())
                .unwrap_or_default();
            out.push(Stat {
                value,
                unit: "date".into(),
                phrase: ctx(&sent, m.start(), m.end()),
                context_sentence: sent.clone(),
                stat_type: "date".into(),
            });
        }
        for caps in duration_re().captures_iter(&sent) {
            let m = caps.get(0).expect("match");
            let unit_str = caps["u"].to_lowercase();
            out.push(Stat {
                value: format!("{} {}", &caps["v"], unit_str),
                unit: unit_str.trim_end_matches('s').to_string(),
                phrase: ctx(&sent, m.start(), m.end()),
                context_sentence: sent.clone(),
                stat_type: "duration".into(),
            });
        }
        for caps in count_re().captures_iter(&sent) {
            let m = caps.get(0).expect("match");
            out.push(Stat {
                value: caps["v"].to_string(),
                unit: caps["u"].to_lowercase(),
                phrase: ctx(&sent, m.start(), m.end()),
                context_sentence: sent.clone(),
                stat_type: "count".into(),
            });
        }
    }
    out
}

#[derive(Debug, Clone, Default, PartialEq)]
pub struct Metadata {
    pub dates: Vec<String>,
    pub amounts: Vec<String>,
    pub urls: Vec<String>,
    pub entities: Vec<String>,  // always empty; Python-only via skimr[ner]

    RE.get_or_init(|| {

            .expect("static regex")
    })
}

    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| {

            r"(?i)\$\d[\d,]*(?:\.\d+)?[KMB]?|",

    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| {
        Regex::new(r#"(?i)https?://[^\s<>"')]+"#).expect("static regex")
    })
}

fn collect_unique(re: &Regex, text: &str) -> Vec<String> {
    let mut seen: std::collections::BTreeSet<String> = std::collections::BTreeSet::new();

        let v = m.as_str().to_string();
        if seen.insert(v.clone()) {
            out.push(v);
        }
    }
    out
}

    Metadata {
        dates: collect_unique(date_re(), text),
        amounts: collect_unique(amount_re(), text),
        urls: collect_unique(url_re(), text),
        entities: Vec::new(),

@lru_cache(maxsize=1)
def _nlp():
    """Return the cached spaCy nlp object. Raises ImportError if not installed."""
    import spacy  # type: ignore[import-not-found]

        return spacy.load("en_core_web_sm")

### Step 3: Implement `src/skimr/extract/phrases.py`

> **ANNOTATION (post-implementation, 2026-04-22):** The literal `_runs()` code below emits **one phrase per stopword-delimited window**. With that logic plus `[a-z]{3,}` tokenization, the tests in Step 1 cannot pass — `"customer support"` (2 tokens) is never a full window; the actual window for that stretch is `"customer support team"` (3 tokens) or `"customer support team evaluated"` (4 tokens). Test 1 would yield empty output.
>
> The shipped implementation at `src/skimr/extract/phrases.py` (commit `b003552`) resolves this by emitting **all contiguous 2–5 token n-grams** per non-stopword run, not just the full run. The same change is applied identically in the Rust port (`rust/src/extract/phrases.rs`), so byte-identical Python↔Rust parity is preserved on shared inputs. Additionally, the implementation uses the `backend=` registry pattern established by T9a — `_regex_phrases` self-registers as `("regex", "phrases")` and the public `phrases()` dispatches through `skimr.extract._backends`.
>
> The plan text below is kept as-is for historical continuity. Treat the tests in Step 1 as the authoritative spec, not the literal `_runs()` body.

```python
"""Heuristic phrase extractor. Repeated 2-5 token runs between stopwords."""
import re

_STOP = set(
    "the a an and or but if then with by of to in on at for from as is are was "
    "were be been being has have had do does did will would could should may "
    "might can must this that these those it its they them their there here "
    "about up down into out over off just also not no our we you your he she "
    "his her him over under more most less least all any some both each every "
    "one two three per".split()
)


def _tokenize_keep_sep(text: str) -> list[str]:
    """Tokenize: words + spans of non-word separators, preserving order."""
    return re.findall(r"\w+|\W+", text.lower())


def _runs(text: str) -> list[str]:
    """Return non-stopword runs of length 2-5 (lowercased)."""
    words = re.findall(r"[a-z]{3,}", text.lower())
    runs: list[str] = []
    buf: list[str] = []
    for w in words:
        if w in _STOP:
            if 2 <= len(buf) <= 5:
                runs.append(" ".join(buf))
            buf = []
        else:
            buf.append(w)
    if 2 <= len(buf) <= 5:
        runs.append(" ".join(buf))
    return runs


def phrases(text: str, keywords: str | None = None) -> tuple[str, ...]:
    """Extract repeated multi-word phrases (2-5 tokens, non-stopword runs)."""
    if not text:
        return ()
    runs = _runs(text)
    counts: dict[str, int] = {}
    order: list[str] = []
    for r in runs:
        counts[r] = counts.get(r, 0) + 1
        if r not in order:
            order.append(r)
    out = [r for r in order if counts[r] >= 2]
    if keywords:
        kw_set = {k.lower() for k in re.findall(r"[a-z]{3,}", keywords.lower())}
        for r in order:
            if counts.get(r, 0) == 1 and any(k in r.split() for k in kw_set):
                out.append(r)
    # Dedupe keeping first-appearance order
    seen: set[str] = set()
    deduped: list[str] = []
    for r in out:
        if r not in seen:
            seen.add(r)
            deduped.append(r)
    return tuple(deduped)
```

//! Phrase extractor — mirrors src/skimr/extract/phrases.py.

use regex::Regex;
use std::collections::BTreeSet;
use std::sync::OnceLock;

fn word_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"[a-z]{3,}").expect("static regex"))
}

fn stop() -> &'static BTreeSet<&'static str> {
    static S: OnceLock<BTreeSet<&'static str>> = OnceLock::new();
    S.get_or_init(|| {
        [
            "the", "a", "an", "and", "or", "but", "if", "then", "with", "by",
            "of", "to", "in", "on", "at", "for", "from", "as", "is", "are",
            "was", "were", "be", "been", "being", "has", "have", "had", "do",
            "does", "did", "will", "would", "could", "should", "may", "might",
            "can", "must", "this", "that", "these", "those", "it", "its",
            "they", "them", "their", "there", "here", "about", "up", "down",
            "into", "out", "over", "off", "just", "also", "not", "no", "our",
            "we", "you", "your", "he", "she", "his", "her", "him", "under",
            "more", "most", "less", "least", "all", "any", "some", "both",
            "each", "every", "one", "two", "three", "per",
        ]
        .into_iter()
        .collect()
    })
}

fn runs(text: &str) -> Vec<String> {
    let lower = text.to_lowercase();
    let mut runs_out: Vec<String> = Vec::new();
    let mut buf: Vec<String> = Vec::new();
    for m in word_re().find_iter(&lower) {
        let w = m.as_str();
        if stop().contains(w) {
            if (2..=5).contains(&buf.len()) {
                runs_out.push(buf.join(" "));
            }
            buf.clear();
        } else {
            buf.push(w.to_string());
        }
    }
    if (2..=5).contains(&buf.len()) {
        runs_out.push(buf.join(" "));
    }
    runs_out
}

#[must_use]
pub fn phrases(text: &str, keywords: Option<&str>) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    let r = runs(text);
    let mut counts: std::collections::HashMap<String, usize> = std::collections::HashMap::new();
    let mut order: Vec<String> = Vec::new();
    for phrase in &r {
        let e = counts.entry(phrase.clone()).or_insert(0);
        if *e == 0 {
            order.push(phrase.clone());
        }
        *e += 1;
    }
    let mut out: Vec<String> = order.iter().filter(|p| counts[*p] >= 2).cloned().collect();
    if let Some(kws) = keywords {
        let kw_set: BTreeSet<String> = word_re()
            .find_iter(&kws.to_lowercase())
            .map(|m| m.as_str().to_string())
            .collect();
        for phrase in &order {
            if counts[phrase] == 1
                && phrase.split_whitespace().any(|w| kw_set.contains(w))
            {
                out.push(phrase.clone());
            }
        }
    }
    // dedupe, preserve order
    let mut seen: BTreeSet<String> = BTreeSet::new();
    out.retain(|p| seen.insert(p.clone()));
    out
}
```

"""extract.correlate_facts — pair repeated entities with nearby numbers."""
import re
from .._types import PhraseFact
from .stats import stats as _stats
from .phrases import phrases as _phrases


_GROWTH_WORDS = {"grew", "grow", "increased", "increase", "rose", "up", "higher", "gained", "added"}
_DECLINE_WORDS = {"fell", "fall", "declined", "decline", "decreased", "decrease", "dropped", "down", "lower", "lost"}


def _polarity(sentence: str) -> str:
    """Infer polarity from cue words in the sentence."""
    lower = sentence.lower()
    toks = set(re.findall(r"[a-z]+", lower))
    if toks & _GROWTH_WORDS:
        return "growth"
    if toks & _DECLINE_WORDS:
        return "decline"
    return "absolute"


def correlate_facts(text: str) -> tuple[PhraseFact, ...]:
    """Return repeated-entity + numeric-fact pairings."""
    if not text:
        return ()
    stats_list = _stats(text)
    if not stats_list:
        return ()

    # Build phrase frequencies — use single-word repeated entities plus phrases
    single_word_counts: dict[str, int] = {}
    for w in re.findall(r"[a-zA-Z]{3,}", text.lower()):
        single_word_counts[w] = single_word_counts.get(w, 0) + 1
    repeated_words = {w for w, c in single_word_counts.items() if c >= 2}

    # Collect multi-word phrases too (already deduped)
    repeated_phrases = set(_phrases(text))

    out: list[PhraseFact] = []
    # For each stat, find the most-frequent repeated entity in the same sentence
    for st in stats_list:
        sent_lower = st.context_sentence.lower()
        # Candidate entities: words in sentence that are repeated AND not stopwords/numbers
        candidates = [
            w for w in re.findall(r"[a-zA-Z]{3,}", sent_lower)
            if w in repeated_words
        ]
        # Prefer a multi-word phrase that appears fully in the sentence
        phrase_match = next(
            (p for p in repeated_phrases if all(w in sent_lower for w in p.split())),
            None,
        )
        entity = phrase_match if phrase_match else (
            max(candidates, key=lambda w: single_word_counts[w]) if candidates else None
        )
        if not entity:
            continue
        out.append(PhraseFact(
            entity=entity,
            number=st.value,
            polarity=_polarity(st.context_sentence),
            sentence=st.context_sentence,
        ))

    # Filter: require each entity to appear with >= 2 distinct facts
    from collections import Counter
    entity_counts = Counter(pf.entity for pf in out)
    return tuple(pf for pf in out if entity_counts[pf.entity] >= 2)
```

//! Pair repeated entities with numeric facts. Mirrors correlate.py.

use crate::extract::phrases::phrases;
use crate::extract::stats::{stats, Stat};
use regex::Regex;
use std::collections::{BTreeSet, HashMap};
use std::sync::OnceLock;

#[derive(Debug, Clone, PartialEq)]
pub struct PhraseFact {
    pub entity: String,
    pub number: String,
    pub polarity: String,
    pub sentence: String,
}

fn word_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"[a-zA-Z]{3,}").expect("static regex"))
}

fn growth_words() -> &'static BTreeSet<&'static str> {
    static S: OnceLock<BTreeSet<&'static str>> = OnceLock::new();
    S.get_or_init(|| {
        ["grew", "grow", "increased", "increase", "rose", "up", "higher", "gained", "added"]
            .into_iter()
            .collect()
    })
}

fn decline_words() -> &'static BTreeSet<&'static str> {
    static S: OnceLock<BTreeSet<&'static str>> = OnceLock::new();
    S.get_or_init(|| {
        ["fell", "fall", "declined", "decline", "decreased", "decrease", "dropped", "down", "lower", "lost"]
            .into_iter()
            .collect()
    })
}

fn polarity(sentence: &str) -> &'static str {
    let lower = sentence.to_lowercase();
    let toks: BTreeSet<String> = word_re()
        .find_iter(&lower)
        .map(|m| m.as_str().to_string())
        .collect();
    if toks.iter().any(|t| growth_words().contains(t.as_str())) {
        return "growth";
    }
    if toks.iter().any(|t| decline_words().contains(t.as_str())) {
        return "decline";
    }
    "absolute"
}

#[must_use]
pub fn correlate_facts(text: &str) -> Vec<PhraseFact> {
    if text.is_empty() {
        return Vec::new();
    }
    let stats_list: Vec<Stat> = stats(text);
    if stats_list.is_empty() {
        return Vec::new();
    }

    // Count single-word occurrences
    let lower = text.to_lowercase();
    let mut word_counts: HashMap<String, usize> = HashMap::new();
    for m in word_re().find_iter(&lower) {
        *word_counts.entry(m.as_str().to_string()).or_insert(0) += 1;
    }
    let repeated_words: BTreeSet<String> = word_counts
        .iter()
        .filter(|(_, c)| **c >= 2)
        .map(|(w, _)| w.clone())
        .collect();

    let repeated_phrases: Vec<String> = phrases(text, None);

    let mut pairings: Vec<PhraseFact> = Vec::new();
    for st in &stats_list {
        let sent_lower = st.context_sentence.to_lowercase();
        let candidates: Vec<String> = word_re()
            .find_iter(&sent_lower)
            .map(|m| m.as_str().to_string())
            .filter(|w| repeated_words.contains(w))
            .collect();

        let phrase_match = repeated_phrases
            .iter()
            .find(|p| p.split_whitespace().all(|w| sent_lower.contains(w)))
            .cloned();

        let entity = if let Some(p) = phrase_match {
            Some(p)
        } else {
            candidates
                .iter()
                .max_by_key(|w| word_counts.get(*w).copied().unwrap_or(0))
                .cloned()
        };

        if let Some(e) = entity {
            pairings.push(PhraseFact {
                entity: e,
                number: st.value.clone(),
                polarity: polarity(&st.context_sentence).to_string(),
                sentence: st.context_sentence.clone(),
            });
        }
    }

    // Keep entities with >= 2 facts
    let mut entity_counts: HashMap<String, usize> = HashMap::new();
    for pf in &pairings {
        *entity_counts.entry(pf.entity.clone()).or_insert(0) += 1;
    }
    pairings
        .into_iter()
        .filter(|pf| entity_counts[&pf.entity] >= 2)
        .collect()
}

# Gold-label protocol for skimr.extract.* fixtures

## Why

SC-D requires each extract primitive to hit ≥0.85 recall / ≥0.80
precision against a hand-labeled gold set. This doc captures the labeling
rules so anyone can re-label or extend the corpus without drift.

## Scope

10 corpora under `benchmarks/corpus/*.txt`. Per corpus, 5 gold files
under `fixtures/extract/<primitive>/<corpus>.json`.

## Format

Each gold file is a JSON object. Missing fields default to empty.

### stats

```json
{
  "facts": [
    {
      "value": "$120K",
      "unit": "usd",
      "stat_type": "money",
      "context_hint": "original quote was $120K"
    }
  ]
}
```
`context_hint` is a string that must appear somewhere in the context_sentence of the matched Stat.

### outline

```json
{
  "sections": [
    {"name": "Facts", "depth": 1},
    {"name": "Discussion", "depth": 2}
  ]
}
```
Order matters; `representative_sentence` is not labeled (too noisy to hand-grade). Precision/recall is over `name`.

### metadata

```json
{
  "dates": ["2025-06-14"],
  "amounts": ["$120K"],
  "urls": [],
  "entities": ["Sarah Jones", "Johnson Education Co"]
}
```
`entities` labeled only when a human would reasonably extract them. Evaluated separately (NER-only path).

### phrases

```json
{"phrases": ["deployment pipeline", "customer support", "action items"]}
```

### correlate

```json
{
  "pairings": [
    {"entity": "revenue", "polarity": "growth"},
    {"entity": "revenue", "polarity": "absolute"}
  ]
}
```
Only `entity` + `polarity` graded; number value is covered by stats.

## Labeling rules

1. Read the source corpus slowly. Mark what a careful human would extract.
2. For stats and metadata: exact string matches preferred; `context_hint` substring is the fallback.
3. For outline: use the section names as they appear in the source, with colons stripped.
4. For phrases: only include phrases that appear at least twice in the source OR contain a clearly-load-bearing term.
5. For correlate: only entities that appear in at least two distinct numeric facts.

## Iteration

Labels may be revised if the eval (Task 13) surfaces patterns the labeler missed — revise the JSON, re-run the eval, commit both.

### Step 1: Implement the harness

```python
# benchmarks/extraction_eval.py
"""Precision/recall eval for skimr.extract primitives vs hand-labeled gold.

Run:
    .venv/bin/python benchmarks/extraction_eval.py

Writes benchmarks/quality/extraction-{date}.md with per-corpus and
aggregate precision / recall / F1 for each of the 5 primitives.
"""
from __future__ import annotations
import json
import sys
import time
from dataclasses import asdict
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / "src"))

from skimr.extract import stats, outline, metadata, phrases, correlate_facts  # noqa: E402

CORPUS_DIR = ROOT / "benchmarks" / "corpus"
GOLD_DIR = ROOT / "fixtures" / "extract"
OUT_DIR = ROOT / "benchmarks" / "quality"
OUT_DIR.mkdir(parents=True, exist_ok=True)


def _prf(tp: int, fp: int, fn: int) -> tuple[float, float, float]:
    p = tp / (tp + fp) if (tp + fp) else 0.0
    r = tp / (tp + fn) if (tp + fn) else 0.0
    f = 2 * p * r / (p + r) if (p + r) else 0.0
    return p, r, f


def eval_stats(corpus_text: str, gold: dict) -> tuple[int, int, int]:
    predicted = stats(corpus_text)
    gold_facts = gold.get("facts", [])
    matched = set()
    tp = 0
    for g in gold_facts:
        hint = g.get("context_hint", "")
        matched_idx = None
        for i, s in enumerate(predicted):
            if i in matched:
                continue
            if (s.stat_type == g.get("stat_type") and
                    hint.lower() in s.context_sentence.lower() and
                    g["value"].lower() in s.value.lower()):
                matched_idx = i
                break
        if matched_idx is not None:
            matched.add(matched_idx)
            tp += 1
    fp = len(predicted) - len(matched)
    fn = len(gold_facts) - tp
    return tp, max(fp, 0), max(fn, 0)


def eval_outline(corpus_text: str, gold: dict) -> tuple[int, int, int]:
    predicted_names = [s.name.lower() for s in outline(corpus_text)]
    gold_names = [s["name"].lower() for s in gold.get("sections", [])]
    tp = sum(1 for n in gold_names if n in predicted_names)
    fp = len(predicted_names) - tp
    fn = len(gold_names) - tp
    return tp, max(fp, 0), max(fn, 0)


def eval_metadata(corpus_text: str, gold: dict) -> tuple[int, int, int]:
    m = metadata(corpus_text)
    tp = fp = fn = 0
    for field in ("dates", "amounts", "urls"):
        pred = set(getattr(m, field))
        g = set(gold.get(field, []))
        tp += len(pred & g)
        fp += len(pred - g)
        fn += len(g - pred)
    return tp, fp, fn


def eval_phrases(corpus_text: str, gold: dict) -> tuple[int, int, int]:
    predicted = set(p.lower() for p in phrases(corpus_text))
    g = set(p.lower() for p in gold.get("phrases", []))
    tp = len(predicted & g)
    fp = len(predicted - g)
    fn = len(g - predicted)
    return tp, fp, fn


def eval_correlate(corpus_text: str, gold: dict) -> tuple[int, int, int]:
    predicted = {(pf.entity.lower(), pf.polarity) for pf in correlate_facts(corpus_text)}
    g = {(pf["entity"].lower(), pf["polarity"]) for pf in gold.get("pairings", [])}
    tp = len(predicted & g)
    fp = len(predicted - g)
    fn = len(g - predicted)
    return tp, fp, fn


EVALS = {
    "stats": eval_stats,
    "outline": eval_outline,
    "metadata": eval_metadata,
    "phrases": eval_phrases,
    "correlate": eval_correlate,
}


def main() -> int:
    date = time.strftime("%Y-%m-%d")
    corpora = sorted(CORPUS_DIR.glob("*.txt"))

    rows: list[dict] = []
    for primitive, fn in EVALS.items():
        agg_tp = agg_fp = agg_fn = 0
        per_corpus: list[tuple[str, int, int, int]] = []
        for p in corpora:
            gold_file = GOLD_DIR / primitive / f"{p.stem}.json"
            gold = json.loads(gold_file.read_text()) if gold_file.exists() else {}
            tp, fp, fn_count = fn(p.read_text(), gold)
            per_corpus.append((p.stem, tp, fp, fn_count))
            agg_tp += tp
            agg_fp += fp
            agg_fn += fn_count
        prec, rec, f1 = _prf(agg_tp, agg_fp, agg_fn)
        rows.append({
            "primitive": primitive,
            "tp": agg_tp, "fp": agg_fp, "fn": agg_fn,
            "precision": prec, "recall": rec, "f1": f1,
            "per_corpus": per_corpus,
        })

    md = [f"# extract.* eval vs gold fixtures — {date}\n\n"]
    md.append("## Aggregate (target: recall >= 0.85, precision >= 0.80)\n\n")
    md.append("| primitive | precision | recall | F1 | TP | FP | FN |\n")
    md.append("|---|---|---|---|---|---|---|\n")
    for row in rows:
        passing = "pass" if row["recall"] >= 0.85 and row["precision"] >= 0.80 else "FAIL"
        md.append(
            f"| `{row['primitive']}` | {row['precision']:.3f} | "
            f"{row['recall']:.3f} | {row['f1']:.3f} | {row['tp']} | "
            f"{row['fp']} | {row['fn']} | **{passing}** |\n"
        )
    md.append("\n## Per-corpus breakdown\n\n")
    for row in rows:
        md.append(f"### `{row['primitive']}`\n\n")
        md.append("| corpus | TP | FP | FN |\n|---|---|---|---|\n")
        for (name, tp, fp, fn_count) in row["per_corpus"]:
            md.append(f"| `{name}` | {tp} | {fp} | {fn_count} |\n")
        md.append("\n")

    out_path = OUT_DIR / f"extraction-{date}.md"
    out_path.write_text("".join(md))
    print(f"Wrote {out_path}")

    # SC-D gate
    failures = [
        r["primitive"] for r in rows
        if r["recall"] < 0.85 or r["precision"] < 0.80
    ]
    if failures:
        print(f"SC-D FAIL for: {failures}")
        return 1
    print("SC-D pass")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

# benchmarks/matrix_eval.py
"""Comparison matrix for skimr v0.2 vs sumy.

Axes:
  - method  (skimr mode+attach, sumy backend)
  - chars   (output length)
  - time    (p50 ms over N iterations, warm)
  - extra   (enrichment counts: stats / outline / metadata / phrases / correlated_facts)
  - A1      (rubric, re-scored or loaded from review doc)
  - A4      (Qwen judge rank aggregate)

Writes benchmarks/quality/matrix-{date}.md.
"""
from __future__ import annotations
import json
import statistics
import subprocess
import sys
import time
from pathlib import Path
from typing import Callable

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT / "src"))

from skimr import summarize  # noqa: E402
from skimr.extract import warmup_ner  # noqa: E402
from sumy.parsers.plaintext import PlaintextParser  # noqa: E402
from sumy.nlp.tokenizers import Tokenizer  # noqa: E402
from sumy.summarizers.lex_rank import LexRankSummarizer  # noqa: E402
from sumy.summarizers.text_rank import TextRankSummarizer  # noqa: E402
from sumy.summarizers.lsa import LsaSummarizer  # noqa: E402

CORPUS_DIR = ROOT / "benchmarks" / "corpus"
OUT_DIR = ROOT / "benchmarks" / "quality"
OUT_DIR.mkdir(parents=True, exist_ok=True)

ITERATIONS = 50
TARGET_CHARS = 500
TARGET_SENTENCES = 3
LATENCY_BUDGET_MS = 250.0


def _timed(fn, text: str) -> tuple[float, object]:
    """Return (p50_ms, last_result). Warmup: 1 unrecorded call."""
    _ = fn(text)
    samples: list[float] = []
    last = None
    for _ in range(ITERATIONS):
        t0 = time.perf_counter_ns()
        last = fn(text)
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    return statistics.median(samples), last


def _skimr_rich(attach: list[str]):
    def run(text):
        return summarize(text, max_length=TARGET_CHARS, attach=attach)
    return run


def _skimr_bare(mode: str):
    def run(text):
        return summarize(text, max_length=TARGET_CHARS, mode=mode)
    return run


def _sumy(cls):
    def run(text):
        parser = PlaintextParser.from_string(text, Tokenizer("english"))
        return " ".join(str(s) for s in cls()(parser.document, TARGET_SENTENCES))
    return run


METHODS: dict[str, Callable[[str], object]] = {
    "skimr/tfidf mode=legacy": _skimr_bare("legacy"),
    "skimr/tfidf mode=default": _skimr_bare("default"),
    "skimr/tfidf mode=default +stats": _skimr_rich(["stats"]),
    "skimr/tfidf mode=default +outline": _skimr_rich(["outline"]),
    "skimr/tfidf mode=default +all": _skimr_rich(
        ["stats", "outline", "metadata", "phrases", "correlated_facts"]),
    "skimr/tfidf mode=coverage": _skimr_bare("coverage"),
    "skimr/tfidf mode=coverage +all": lambda t: summarize(
        t, max_length=TARGET_CHARS, mode="coverage",
        attach=["stats", "outline", "metadata", "phrases", "correlated_facts"],
    ),
    "sumy/LexRank": _sumy(LexRankSummarizer),
    "sumy/TextRank": _sumy(TextRankSummarizer),
    "sumy/LSA": _sumy(LsaSummarizer),
}


def _rust_tfidf_bench(text: str) -> tuple[float, int]:
    """Call rust/target/release/bench_tfidf for a single corpus. Returns (p50_ms, chars)."""
    bin_path = ROOT / "rust" / "target" / "release" / "bench_tfidf"
    if not bin_path.exists():
        return (-1.0, 0)
    proc = subprocess.run(
        [str(bin_path), str(ITERATIONS)],
        input=text, capture_output=True, text=True, check=True,
    )
    data = json.loads(proc.stdout.strip())
    return data["p50_ms"], data["out_chars"]


def _count_extras(result) -> dict[str, int]:
    from skimr import SummaryResult
    if not isinstance(result, SummaryResult):
        return {"stats": 0, "outline": 0, "metadata": 0, "phrases": 0, "correlated_facts": 0}
    return {
        "stats": len(result.stats or ()),
        "outline": len(result.outline or ()),
        "metadata": (
            len((result.metadata.dates if result.metadata else ())) +
            len((result.metadata.amounts if result.metadata else ())) +
            len((result.metadata.urls if result.metadata else ())) +
            len((result.metadata.entities if result.metadata else ()))
        ) if result.metadata else 0,
        "phrases": len(result.phrases or ()),
        "correlated_facts": len(result.correlated_facts or ()),
    }


def _summary_text(result) -> str:
    return result.summary if hasattr(result, "summary") else str(result)


def main() -> int:
    warmup_ner()  # load spaCy before any timed call
    date = time.strftime("%Y-%m-%d")
    corpora = sorted(CORPUS_DIR.glob("*.txt"))
    rows: list[dict] = []

    for method, fn in METHODS.items():
        method_rows: list[dict] = []
        for p in corpora:
            text = p.read_text()
            p50, last = _timed(fn, text)
            extras = _count_extras(last)
            chars = len(_summary_text(last))
            method_rows.append({
                "corpus": p.stem, "p50_ms": p50, "chars": chars, "extras": extras,
            })
        rows.append({"method": method, "corpus_rows": method_rows})

    # Rust benchmark via bench_tfidf helper
    rust_rows: list[dict] = []
    for p in corpora:
        text = p.read_text()
        p50, chars = _rust_tfidf_bench(text)
        rust_rows.append({
            "corpus": p.stem, "p50_ms": p50, "chars": chars,
            "extras": {"stats": 0, "outline": 0, "metadata": 0, "phrases": 0, "correlated_facts": 0},
        })
    rows.append({"method": "rust-skimr/tfidf mode=default", "corpus_rows": rust_rows})

    md: list[str] = [f"# Comparison matrix — skimr v0.2 vs sumy ({date})\n\n"]
    md.append(f"**Iterations per cell:** {ITERATIONS}. **Budget gate:** p50 ≤ {LATENCY_BUDGET_MS} ms on every corpus.\n\n")
    md.append("## Aggregate by method (averaged across 10 corpora)\n\n")
    md.append("| method | avg chars | avg p50 ms | avg stats | avg outline | avg metadata | avg phrases | avg correlated | budget? |\n")
    md.append("|---|---|---|---|---|---|---|---|---|\n")
    for r in rows:
        cr = r["corpus_rows"]
        avg_chars = statistics.mean(x["chars"] for x in cr)
        avg_p50 = statistics.mean(x["p50_ms"] for x in cr if x["p50_ms"] >= 0)
        avg_ex = {k: statistics.mean(x["extras"][k] for x in cr) for k in cr[0]["extras"]}
        max_p50 = max(x["p50_ms"] for x in cr if x["p50_ms"] >= 0)
        budget_ok = "pass" if max_p50 <= LATENCY_BUDGET_MS else "**FAIL**"
        md.append(
            f"| `{r['method']}` | {avg_chars:.0f} | {avg_p50:.2f} | "
            f"{avg_ex['stats']:.1f} | {avg_ex['outline']:.1f} | "
            f"{avg_ex['metadata']:.1f} | {avg_ex['phrases']:.1f} | "
            f"{avg_ex['correlated_facts']:.1f} | {budget_ok} |\n"
        )

    md.append("\n## Per-corpus latency (ms p50)\n\n")
    methods_hdr = [r["method"] for r in rows]
    md.append("| corpus | " + " | ".join(f"`{m}`" for m in methods_hdr) + " |\n")
    md.append("|" + "|".join(["---"] * (1 + len(methods_hdr))) + "|\n")
    corpus_names = [p.stem for p in corpora]
    for cname in corpus_names:
        row_cells = [f"`{cname}`"]
        for r in rows:
            cell = next((x for x in r["corpus_rows"] if x["corpus"] == cname), None)
            row_cells.append(f"{cell['p50_ms']:.2f}" if cell else "—")
        md.append("| " + " | ".join(row_cells) + " |\n")

    out_path = OUT_DIR / f"matrix-{date}.md"
    out_path.write_text("".join(md))
    print(f"Wrote {out_path}")

    # SC-B gate
    over_budget = [
        (r["method"], c["corpus"], c["p50_ms"])
        for r in rows for c in r["corpus_rows"]
        if c["p50_ms"] > LATENCY_BUDGET_MS
    ]
    if over_budget:
        print("SC-B FAIL on:")
        for m, cor, ms in over_budget:
            print(f"  {m} on {cor}: {ms:.1f} ms")
        return 1
